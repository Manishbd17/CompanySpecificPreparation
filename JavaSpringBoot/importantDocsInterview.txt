 public static int getDuplicateCnt(List<Integer> myList) {
 	int dupCnt=0;
 	int n=myList.length();

 public reverse(int[] lst,int st,int end) {
 	while(st<=end) {
 		int temp=lst[st];
 		lst[st]=lst[end];
 		lst[end]=temp;
 		st++;
 		end--;
 	}
 }

 public int[] rotate(int[] lst,int d) {
 	reverse(lst,0,d-1);
 	reverse(lst,d,n-1);
 	reverse(lst,0,n-1);
 	return lst;
 }
