 public static int lastIdx(int[] arr,int target) {
        int n=arr.length;
        int st=0,end=n-1;
        int lastIdx=-1;
        while(st<=end) {
            int mid = st+(end-st)/2;
            if(arr[mid]==target) {
                lastIdx = mid;
                st = mid + 1;
            } else if(arr[mid] < target) {
                st=mid+1;
            } else {
                end = mid-1;
            }
        }
        return lastIdx;
    }
//nums = [5,7,7,8,8,10], target = 8
    public static int firstIdx(int[] arr,int target) {
        int n=arr.length;
        int st=0,end=n-1;
        int firstIdx=-1;
        while(st<=end) {
            int mid = st+(end-st)/2;
            if(arr[mid]==target) {
                firstIdx = mid;
                end = mid-1;
            } else if(arr[mid] < target) {
                st=mid+1;
            } else {
                end = mid-1;
            }
        }
        return firstIdx;
    }

public class Singleton {

    private static Singleton instance = null;

    private Singleton() { };

    public static Singleton getInstance() {
        if(instance == null) {
            synchronized(Singleton.class) {
                if(instance == null) {
                    instance = new Singleton();
                }
            }
        }
        return instance;
    }
}
